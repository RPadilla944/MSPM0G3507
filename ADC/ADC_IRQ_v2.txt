#include "ti/devices/msp/m0p/mspm0g350x.h"
#include "ti/driverlib/dl_adc12.h"
#include "ti/driverlib/dl_common.h" // For DL_Common_delayCycles function
#include "ti_msp_dl_config.h"
#include "stdio.h"
int count=1;
int main (void)
{
 
  ADC0->ULLMEM.GPRCM.RSTCTL = (0xB1 << 24) | ((1 << 1) | (1 << 0)); // reset ADC0 pins, enable, set
  ADC0->ULLMEM.GPRCM.PWREN = (0x26 << 24) | (1 << 0);               // activate ADC0 pins
  DL_Common_delayCycles(40000000);                          // 1/2 sec delay

  ADC0->ULLMEM.GPRCM.CLKCFG = (0xA9 << 24) | (0x0 << 0);            // ULPCLK 0xA9 (bits 31–24)
  ADC0->ULLMEM.CLKFREQ = 7;                                         // 40-48 MHz
  ADC0->ULLMEM.CTL0 = (3 << 16) | (1 << 8);                         // divide by 8
  ADC0->ULLMEM.CTL1 = 0;                                            // mode
  ADC0->ULLMEM.CTL2 = 0;                                            // MEMRES
  ADC0->ULLMEM.MEMCTL[0] = 3;                                       // channel 3 PA24 ADC0pin
  ADC0->ULLMEM.SCOMP0 = 0;                                          // 8 sample clocks
  ADC0->ULLMEM.CPU_INT.ICLR = (1 << 24);                            // clear bit 24
  ADC0->ULLMEM.CPU_INT.IMASK = (1 << 8);                            // Enable MEM0 interrupt  

  NVIC->IP[1] = (NVIC->IP[1]&~(3 << 6))|(2 << 6);                   // set priority (bits 15,14) IRQ 4
  NVIC->ISER[0] = 1 << 4;                                           // ADC0 interrupt

  __enable_irq();

  while(1)
  {
  ADC0->ULLMEM.CTL0 |= 1;                   //  enable conversions
  ADC0->ULLMEM.CTL1 |= (1 << 8);            //  start ADC

  }
    
}
void ADC0_IRQHandler(void) 
{
uint32_t adcRaw= ADC0->ULLMEM.MEMRES[0];
  if(adcRaw>1000)
  {  
    printf("counter = %d\n", count);
    count++;
  }
    ADC0->ULLMEM.CPU_INT.ICLR = (1 << 8);
}

    The digital output would be 63*1V/4V, or 64*1V/4V, which is 16

Formula: bits= N*(input/range)

NOTE: ADC has a peak at 4095

For ADC it is wise not to print anything or add delays in the handler but for the sake of this code we have it to show data is working
